Ерөнхий дундажаас өндөр дүнтэй захиалгууд

v_customer_sales харагдац нь харилцагч бүрийн захиалгын тоо, нийт орлого,
захиалгын дундаж дүн болон сүүлийн захиалгын огноог нэг мөрөөр харуулна.
LEFT JOIN болон COALESCE ашигласнаар захиалгагүй харилцагчийн орлого 0 байна.
Эцсийн асуулга нь захиалгын дундаж нь бүх захиалгын ерөнхий дунджаас өндөр
харилцагчдыг шүүнэ.


SQL асуулга
CREATE OR REPLACE VIEW v_customer_sales AS
SELECT
    c.customer_id,
    c.company_name,
    c.country,
    COUNT(vot.order_id) AS order_count,
    COALESCE(SUM(vot.order_total), 0::numeric) AS total_revenue,
    COALESCE(AVG(vot.order_total), 0::numeric) AS average_order_value,
    MAX(vot.order_date) AS last_order_date
FROM customers AS c
LEFT JOIN v_order_totals AS vot ON vot.customer_id = c.customer_id
GROUP BY c.customer_id, c.company_name, c.country;

SELECT customer_id, company_name, country, order_count,
       ROUND(total_revenue, 2) AS total_revenue,
       ROUND(average_order_value, 2) AS average_order_value,
       last_order_date
FROM v_customer_sales
WHERE average_order_value > (SELECT AVG(order_total) FROM v_order_totals)
ORDER BY average_order_value DESC, customer_id;

Ангилал бүрийн шилдэг борлуулалттай 3
бүтээгдэхүүн

Бүтээгдэхүүн бүрийн борлуулалтын дүнг эхлээд нэгтгэж, дараа нь RANK() функцийг
ангилал тус бүрээр ажиллуулав. Эцсийн асуулга нь sales_rank <= 3 нөхцөлөөр
ангилал бүрийн эхний гурван бүтээгдэхүүнийг харуулна.


SQL асуулга
CREATE OR REPLACE VIEW v_product_category_rank AS
WITH product_sales AS (
    SELECT
        c.category_id,
        c.category_name,
        p.product_id,
        p.product_name,
        COALESCE(
            SUM((od.unit_price * od.quantity * (1 - od.discount))::numeric),
            0::numeric
        ) AS total_sales
    FROM categories AS c
    JOIN products AS p ON p.category_id = c.category_id
    LEFT JOIN order_details AS od ON od.product_id = p.product_id
    GROUP BY c.category_id, c.category_name, p.product_id, p.product_name
)
SELECT
    category_id,
    category_name,
    product_id,
    product_name,
    total_sales,
    RANK() OVER (
        PARTITION BY category_id
        ORDER BY total_sales DESC
    ) AS sales_rank
FROM product_sales;

SELECT category_name, product_name,
       ROUND(total_sales, 2) AS total_sales,
       sales_rank
FROM v_product_category_rank
WHERE sales_rank <= 3
ORDER BY category_name, sales_rank, product_name;

Рекурсив CTE ашигласан ажилтны шатлал

Менежергүй ажилтныг суурь мөр болгож, employees.reports_to холбоосоор доод
түвшний ажилтнуудыг рекурсив байдлаар нэмэв. Харагдац нь ажилтан, менежер,
шатлалын гүн болон эх ажилтнаас тухайн ажилтан хүртэлх замыг хадгална.


SQL асуулга
CREATE OR REPLACE VIEW v_employee_hierarchy AS
WITH RECURSIVE hierarchy AS (
    -- Root employee: no manager
    SELECT
        e.employee_id,
        CONCAT(e.first_name, ' ', e.last_name) AS employee_name,
        e.reports_to AS manager_id,
        NULL::text AS manager_name,
        0 AS depth,
        CONCAT(e.first_name, ' ', e.last_name)::text AS path
    FROM employees AS e
    WHERE e.reports_to IS NULL

    UNION ALL

    -- Add employees who report to the current level
    SELECT
        e.employee_id,
        CONCAT(e.first_name, ' ', e.last_name) AS employee_name,
        e.reports_to AS manager_id,
        h.employee_name AS manager_name,
        h.depth + 1 AS depth,
        (h.path || ' > ' || CONCAT(e.first_name, ' ', e.last_name))::text AS path
    FROM employees AS e
    JOIN hierarchy AS h ON e.reports_to = h.employee_id
)
SELECT employee_id, employee_name, manager_id, manager_name, depth, path
FROM hierarchy;

SELECT *
FROM v_employee_hierarchy
ORDER BY depth, employee_id;

Материалчлагдсан сарын борлуулалтын харагдац

mv_monthly_sales материалчлагдсан харагдац нь улс болон жил-сараар нийт
орлого, захиалгын тоог урьдчилан тооцож хадгална. year_month, country хос
баганад unique индекс үүсгэснээр хэрэглэгчдийн уншилтыг удаан хаахгүй REFRESH
MATERIALIZED VIEW CONCURRENTLY командыг ажиллуулах боломжтой болсон.

Энгийн харагдац эх хүснэгт бүрийг уншихдаа дахин тооцдог. Харин энэ тайлан
олон давтамжтай уншигдаж, захиалгын өгөгдөл харьцангуй цөөн шинэчлэгддэг үед
материалчлагдсан харагдац илүү хурдан байна. Сул тал нь REFRESH хийх хүртэл
хуучин үр дүн харуулж болно.


SQL асуулга
DROP MATERIALIZED VIEW IF EXISTS mv_monthly_sales;

CREATE MATERIALIZED VIEW mv_monthly_sales AS
SELECT
    DATE_TRUNC('month', o.order_date)::date AS year_month,
    o.ship_country AS country,
    COUNT(DISTINCT o.order_id) AS order_count,
    SUM((od.unit_price * od.quantity * (1 - od.discount))::numeric) AS total_revenue
FROM orders AS o
JOIN order_details AS od ON od.order_id = o.order_id
WHERE o.order_date IS NOT NULL
  AND o.ship_country IS NOT NULL
GROUP BY DATE_TRUNC('month', o.order_date)::date, o.ship_country;

CREATE UNIQUE INDEX ux_mv_monthly_sales_year_month_country
    ON mv_monthly_sales (year_month, country);

-- The view already contains data, and the unique index makes this possible.
REFRESH MATERIALIZED VIEW CONCURRENTLY mv_monthly_sales;

SELECT year_month, country, order_count,
       ROUND(total_revenue, 2) AS total_revenue
FROM mv_monthly_sales
ORDER BY year_month, country;

-- A materialized view is useful when this report is read often but source
-- orders change less often. Reads are faster because the aggregate is stored.
-- Unlike a normal view, it can be out of date until REFRESH is executed.

Харагдац ашигласан харагдац ба сарын өөрчлөлт

Эхний v_order_lines харагдац захиалгын мөр бүрийн дүнг тооцно.      Үүнээс
v_category_monthly нь ангилал тус бүрийн сарын орлогыг нэгтгэж, гурав дахь
v_category_monthly_change харагдац LAG() ашиглан өмнөх сарын орлогыг авна.
Өөрчлөлтийн дүнгээр өсөхөөр эрэмбэлж эхний мөрийг сонгосноор хамгийн их

SQL асуулга
CREATE OR REPLACE VIEW v_order_lines AS
SELECT
    o.order_id,
    o.order_date,
    p.product_id,
    p.product_name,
    c.category_id,
    c.category_name,
    od.unit_price,
    od.quantity,
    od.discount,
    (od.unit_price * od.quantity * (1 - od.discount))::numeric AS line_total
FROM orders AS o
JOIN order_details AS od ON od.order_id = o.order_id
JOIN products AS p ON p.product_id = od.product_id
JOIN categories AS c ON c.category_id = p.category_id;

CREATE OR REPLACE VIEW v_category_monthly AS
SELECT
    category_id,
    category_name,
    DATE_TRUNC('month', order_date)::date AS month,
    SUM(line_total) AS current_revenue
FROM v_order_lines
GROUP BY category_id, category_name, DATE_TRUNC('month', order_date)::date;

CREATE OR REPLACE VIEW v_category_monthly_change AS
WITH monthly_with_previous AS (
    SELECT
        category_id,
        category_name,
        month,
        current_revenue,
        LAG(current_revenue) OVER (
            PARTITION BY category_id
            ORDER BY month
        ) AS previous_revenue
    FROM v_category_monthly
)
SELECT
    category_id,
    category_name,

    month,
    previous_revenue,
    current_revenue,
    current_revenue - previous_revenue AS change_amount,
    100 * (current_revenue - previous_revenue)
        / NULLIF(previous_revenue, 0) AS change_percent
FROM monthly_with_previous;

-- The category with the largest one-month decrease.
SELECT
    category_name,
    month,
    ROUND(previous_revenue, 2) AS previous_revenue,
    ROUND(current_revenue, 2) AS current_revenue,
    ROUND(change_amount, 2) AS change_amount,
    ROUND(change_percent, 2) AS change_percent
FROM v_category_monthly_change
WHERE previous_revenue IS NOT NULL
ORDER BY change_amount ASC
LIMIT 1;

Хожимдсон хүргэлтийн шинжилгээ

CASE илэрхийлэл нь хүргэлт бүрийг On time, Late, Not shipped төлөвт ангилна.
Хүргэсэн захиалгуудыг суурь болгон тээвэрлэгч бүрийн хожимдсон хүргэлтийн хувь
болон зөвхөн хоцорсон хүргэлтийн дундаж өдрийг тооцоолж, хамгийн муугаас
хамгийн сайн руу эрэмбэлэв.


SQL асуулга
CREATE OR REPLACE VIEW v_shipment_performance AS
SELECT
    o.order_id,
    o.required_date,
    o.shipped_date,
    o.shipped_date - o.required_date AS delivery_days,
    CASE
         WHEN o.shipped_date IS NULL THEN 'Not shipped'
         WHEN o.shipped_date <= o.required_date THEN 'On time'
         ELSE 'Late'
    END AS shipment_status,
    s.company_name AS shipper_name
FROM orders AS o
JOIN shippers AS s ON s.shipper_id = o.ship_via;

SELECT
    shipper_name,
    COUNT(*) FILTER (WHERE shipped_date IS NOT NULL) AS shipped_orders,
    COUNT(*) FILTER (WHERE shipment_status = 'Late') AS late_orders,
    ROUND(
        100.0 * COUNT(*) FILTER (WHERE shipment_status = 'Late')
        / NULLIF(COUNT(*) FILTER (WHERE shipped_date IS NOT NULL), 0),
        2
    ) AS late_percentage,
    ROUND(
        AVG(delivery_days) FILTER (WHERE shipment_status = 'Late'),
        2
    ) AS average_late_days
FROM v_shipment_performance
GROUP BY shipper_name
ORDER BY late_percentage DESC, average_late_days DESC;

Бараа материалын төлөвийн харагдац

v_product_inventory_status харагдац нь бүтээгдэхүүн, нийлүүлэгч, ангиллын
мэдээллийг нэгтгэж нөөцийн үнийг тооцно. CASE логикоор захиалгад байгаа хэмжээг
оролцуулсан ч дахин захиалах түвшинд хүрэхгүй бол Reorder needed, одоогийн
үлдэгдэл бага бол Low, бусад үед OK төлөв өгнө.      Захиалгад орохоо больсон
бүтээгдэхүүнийг оруулаагүй.


SQL асуулга
CREATE OR REPLACE VIEW v_product_inventory_status AS
SELECT
    p.product_id,
    p.product_name,
    s.supplier_id,
    s.company_name AS supplier_name,
    c.category_name,
    p.unit_price,
    p.units_in_stock,
    p.units_on_order,
    p.reorder_level,
    (p.unit_price * p.units_in_stock)::numeric AS stock_value,
    CASE
         WHEN p.units_in_stock + p.units_on_order < p.reorder_level
             THEN 'Reorder needed'
         WHEN p.units_in_stock < p.reorder_level
             THEN 'Low'
         ELSE 'OK'
    END AS stock_status
FROM products AS p
JOIN suppliers AS s ON s.supplier_id = p.supplier_id
JOIN categories AS c ON c.category_id = p.category_id
WHERE p.discontinued = 0;

-- View result.
SELECT
    product_name,
    supplier_name,
    category_name,
    unit_price,
    units_in_stock,
    units_on_order,
    reorder_level,
    ROUND(stock_value, 2) AS stock_value,
    stock_status
FROM v_product_inventory_status
ORDER BY supplier_name, product_name;

-- Supplier summary for products that need reordering.
SELECT
    supplier_name,
    COUNT(*) FILTER (WHERE stock_status = 'Reorder needed') AS reorder_products,
    ROUND(
        SUM(stock_value) FILTER (WHERE stock_status = 'Reorder needed'),
        2
    ) AS total_stock_value
FROM v_product_inventory_status
GROUP BY supplier_id, supplier_name
HAVING COUNT(*) FILTER (WHERE stock_status = 'Reorder needed') > 0
ORDER BY reorder_products DESC, supplier_name;
